# Global blackboard entries shared between the behavior mode executor (which writes them) and the SendVehicleCommand
# node (which reads them). The '@' prefix denotes the root (global) blackboard. Applied as compile definitions so both
# sides use the exact same key string (single source of truth).
# - SOURCE_COMPONENT: the in-charge BehaviorModeExecutor's VehicleCommand source component (for command attribution).
# - MODE_EXECUTOR_ACTIVE: boolean flag set while a BehaviorModeExecutor is in charge. When set, SendVehicleCommand
#   publishes to the mode-executor command topic instead of the default external one.
set(AUTO_APMS_PX4_SOURCE_COMPONENT_GLOBAL_KEY "@_px4_source_component")
set(AUTO_APMS_PX4_MODE_EXECUTOR_ACTIVE_GLOBAL_KEY "@_px4_mode_executor_active")

#
# ------------ Behavior Mode Executor --------------
#
# PX4 mode executor that runs an AutoAPMS behavior in-process when put in charge by the FMU. Configured entirely via
# ROS 2 parameters. The behavior is assembled from standard AutoAPMS nodes (requires the PX4 skills to be active).
# Registered as a ROS 2 component (single node hosting the mode executor, its owned mode and the tree engine).

generate_parameter_library(behavior_mode_executor_params
  "config/behavior_mode_executor_params.yaml"
)

add_library(${PROJECT_NAME}_behavior_mode_executor SHARED
  "src/behavior_mode_executor.cpp"
)
target_include_directories(${PROJECT_NAME}_behavior_mode_executor PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)
target_link_libraries(${PROJECT_NAME}_behavior_mode_executor PUBLIC
  ${PROJECT_NAME}
  behavior_mode_executor_params
  rclcpp::rclcpp
  rclcpp_components::component
  px4_ros2_cpp::px4_ros2_cpp
  Eigen3::Eigen
  auto_apms_util::auto_apms_util
  auto_apms_behavior_tree::auto_apms_behavior_tree
)

target_compile_definitions(${PROJECT_NAME}_behavior_mode_executor PRIVATE
  AUTO_APMS_PX4_SOURCE_COMPONENT_GLOBAL_KEY="${AUTO_APMS_PX4_SOURCE_COMPONENT_GLOBAL_KEY}"
  AUTO_APMS_PX4_MODE_EXECUTOR_ACTIVE_GLOBAL_KEY="${AUTO_APMS_PX4_MODE_EXECUTOR_ACTIVE_GLOBAL_KEY}"
)

rclcpp_components_register_node(${PROJECT_NAME}_behavior_mode_executor
  PLUGIN auto_apms_px4::BehaviorModeExecutorNode
  EXECUTABLE behavior_mode_executor
)

# Give the plugin sources the public SendVehicleCommand header and the AUTO_APMS_PX4_SOURCE_COMPONENT_GLOBAL_KEY
# compile definition directly. We deliberately do NOT link the main ${PROJECT_NAME} target here: it doubles as the
# generated node model header target (see NODE_MODEL_HEADER_TARGET below), which depends back on this plugin and would
# form a target dependency cycle. Hence send_vehicle_command.cpp is compiled into both targets.
target_link_libraries(${PROJECT_NAME}_behavior_tree_nodes
  auto_apms_behavior_tree_core::auto_apms_behavior_tree_core
  px4_ros2_cpp::px4_ros2_cpp
  Eigen3::Eigen
  ${px4_msgs_TARGETS}
  ${auto_apms_px4_interfaces_TARGETS}
)
target_include_directories(${PROJECT_NAME}_behavior_tree_nodes PRIVATE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
)
target_compile_definitions(${PROJECT_NAME}_behavior_tree_nodes PRIVATE
  AUTO_APMS_PX4_SOURCE_COMPONENT_GLOBAL_KEY="${AUTO_APMS_PX4_SOURCE_COMPONENT_GLOBAL_KEY}"
  AUTO_APMS_PX4_MODE_EXECUTOR_ACTIVE_GLOBAL_KEY="${AUTO_APMS_PX4_MODE_EXECUTOR_ACTIVE_GLOBAL_KEY}"
  AUTO_APMS_PX4_EXPORT_NODE_PLUGIN
